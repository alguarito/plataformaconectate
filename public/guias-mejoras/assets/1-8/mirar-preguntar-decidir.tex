% Mirar → preguntar → decidir → (solo ahora) calcular.
%
% Diagrama propio en TikZ para la guía 8-1-1 «Preguntar antes de calcular».
% Sin librerías extra: nodos con esquinas redondeadas y flechas simples.
% Colores de la plantilla: milcVerde (mirar) · milcTurquesa (preguntar) ·
% milcMagenta (decidir) · gris punteado para el cálculo, que va al final.
%
% Las anotaciones de dentro nombran cada paso; la síntesis va en el pie de
% figura (vía \guiaDiagrama), no aquí. Ancho total ≈ 14 cm.
\begin{tikzpicture}[scale=1,font=\sffamily,>=stealth,
  caja/.style={rounded corners=3pt,minimum height=1.05cm,minimum width=2.9cm,
               text=white,font=\sffamily\bfseries\small,inner sep=4pt},
  sub/.style={font=\sffamily\scriptsize\itshape,text=milcNegro!75,text width=3cm,align=center},
  flecha/.style={->,line width=1.1pt,milcNegro}]

  % ── Los tres pasos que van antes de cualquier fórmula ─────────────────
  \node[caja,fill=milcVerde]    (mirar)     at (0,0)    {MIRAR};
  \node[caja,fill=milcTurquesa] (preguntar) at (3.7,0)  {PREGUNTAR};
  \node[caja,fill=milcMagenta]  (decidir)   at (7.4,0)  {DECIDIR};

  % ── El cálculo, en gris y punteado: solo ahora ─────────────────────────
  \node[rounded corners=3pt,draw=milcNegro!50,dashed,line width=0.7pt,
        minimum height=1.05cm,minimum width=2.2cm,
        font=\sffamily\bfseries\small,text=milcNegro!70] (calcular) at (10.9,0) {calcular};

  \draw[flecha] (mirar) -- (preguntar);
  \draw[flecha] (preguntar) -- (decidir);
  \draw[flecha,milcNegro!50] (decidir) -- (calcular);

  % ── Qué hace cada paso ─────────────────────────────────────────────────
  \node[sub] at (0,-0.95)    {qué hay en la hoja};
  \node[sub] at (3.7,-0.95)  {las cinco preguntas};
  \node[sub] at (7.4,-0.95)  {qué calcular y qué no};
  \node[sub] at (10.9,-0.95) {solo ahora};

  % ── La tira de las cinco preguntas, bajo el paso PREGUNTAR ─────────────
  \begin{scope}[shift={(0,-2.15)}]
    \foreach \i/\t in {0/origen, 1/autor y fecha, 2/tipo de dato, 3/errores y vacíos, 4/pregunta}{
      \node[draw=milcTurquesa,line width=0.6pt,rounded corners=2pt,
            minimum height=0.55cm,minimum width=2.15cm,
            font=\sffamily\scriptsize,text=milcNegro,inner sep=2pt]
        at (\i*2.45+0.55,0) {\textbf{\the\numexpr\i+1\relax}\enspace\t};
    }
    \draw[milcTurquesa,line width=0.6pt] (0.55,0.75) -- (0.55,0.42);
    \draw[milcTurquesa,line width=0.6pt] (0.55,0.75) -- (3.7,0.75) -- (3.7,0.95);
  \end{scope}

\end{tikzpicture}

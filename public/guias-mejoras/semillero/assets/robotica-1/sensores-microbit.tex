% Los sensores integrados del micro:bit — con qué percibe el mundo.
%
% Diagrama propio en TikZ (sin licencias de terceros, vectorial, editable). Se
% tiñe con el color de la línea (milcGradeDark = naranja de Robótica). Dibuja el
% micro:bit y señala sus sensores integrados, para mapear «qué necesidad del
% entorno alcanza cada sensor». No requiere cablear nada: todo está adentro.
%
% Se incrusta vía \guiaDiagrama desde el bloque `recursos:` del YAML.
\begin{tikzpicture}[scale=1,font=\sffamily,>=stealth]

  % ── Cuerpo del micro:bit ─────────────────────────────────────────────
  \fill[milcGradeDark] (0,0) rectangle (5.4,4.2);
  \draw[milcNegro!55,line width=.6pt,rounded corners=2pt] (0,0) rectangle (5.4,4.2);

  % Botones A y B
  \foreach \x/\lbl in {0.7/A,4.7/B}{
    \fill[white] (\x,2.9) circle (0.32);
    \draw[milcNegro!55,line width=.5pt] (\x,2.9) circle (0.32);
    \node[font=\scriptsize\bfseries,milcNegro] at (\x,2.9) {\lbl};
  }

  % Matriz de 25 LED (5x5)
  \foreach \i in {0,...,4}{
    \foreach \j in {0,...,4}{
      \fill[milcMostaza] ({1.5+\i*0.6},{0.5+\j*0.45}) rectangle ({1.5+\i*0.6+0.22},{0.5+\j*0.45+0.28});
    }
  }

  % Pines dorados abajo
  \foreach \x in {0.5,1.4,2.7,4.0,4.9}{
    \fill[milcMostaza!85!black] (\x,-0.02) rectangle ({\x+0.35},0.18);
  }

  % ── Rótulos de sensores (callouts) ───────────────────────────────────
  \node[anchor=east,font=\scriptsize\bfseries,milcTurquesa,align=right,text width=3.3cm] at (-0.3,3.6) {sensor de luz\\\mdseries\itshape la matriz de LED también mide la luz};
  \draw[->,milcNegro!55,line width=.5pt] (-0.3,3.5) -- (2.6,1.9);

  \node[anchor=east,font=\scriptsize\bfseries,milcTurquesa,align=right,text width=3.3cm] at (-0.3,1.9) {acelerómetro\\\mdseries\itshape movimiento, agitar, caída, inclinación};
  \draw[->,milcNegro!55,line width=.5pt] (-0.3,1.9) -- (1.5,1.6);

  \node[anchor=east,font=\scriptsize\bfseries,milcTurquesa,align=right,text width=3.3cm] at (-0.3,0.5) {botones A y B\\\mdseries\itshape orden de una persona};
  \draw[->,milcNegro!55,line width=.5pt] (-0.3,0.6) -- (0.4,2.75);

  \node[anchor=west,font=\scriptsize\bfseries,milcTurquesa,align=left,text width=3.3cm] at (5.7,3.6) {sensor de temperatura\\\mdseries\itshape calor del ambiente};
  \draw[->,milcNegro!55,line width=.5pt] (5.7,3.5) -- (3.9,2.4);

  \node[anchor=west,font=\scriptsize\bfseries,milcTurquesa,align=left,text width=3.3cm] at (5.7,1.9) {brújula\\\mdseries\itshape dirección, campo magnético};
  \draw[->,milcNegro!55,line width=.5pt] (5.7,1.9) -- (3.6,1.4);

  \node[anchor=west,font=\scriptsize\bfseries,milcTurquesa,align=left,text width=3.3cm] at (5.7,0.5) {pines\\\mdseries\itshape tacto y sensores externos};
  \draw[->,milcNegro!55,line width=.5pt] (5.7,0.55) -- (2.7,0.08);

  % ── Moraleja ─────────────────────────────────────────────────────────
  \node[font=\scriptsize\itshape,milcNegro!75,align=center,text width=11cm]
    at (2.7,-0.95)
    {El micro:bit ya trae con qué \textbf{percibir} el mundo ---luz, temperatura,
     movimiento, dirección, tacto--- sin cablear nada. La pregunta del
     investigador no es «¿qué sensor tengo?», sino \textbf{«¿qué de mi entorno
     necesita ser percibido?»}.};

\end{tikzpicture}
